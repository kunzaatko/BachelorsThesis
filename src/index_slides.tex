        d \ge \frac{\alt<2->{\textcolor<2>{Orange}{\qty{488}{\nano\meter}}}{\lambda}}{2
        \cdot\alt<2->{\textcolor<2>{Orange}{1.4}}{\NA}}\only<3->{\approx\textcolor<3>{Orange}{\qty{174}{\nano\meter}}}
    \]
\end{frame}

\begin{frame}{How does \acs{SIM} work?}
    \begin{itemize}
            \item \emph{harmonic} illumination intensity instead of uniform \pause

                \ip_{(m,\vec{k}_\ip,\phi)}(\vec{r}) = 1 + \tfrac{m}{2} \textcolor{Orange}{\cos}\ab(2\pi \vec{k}_\ip \cdot \vec{r} + \phi)

\begin{frame}{How does \acs{SIM} work?}
    \only<1,3,5,6,7,8,9,13>{
        \begin{itemize}
            \item \emph{harmonic} illumination intensity instead of uniform
            \only<1-11>{\[
                \ip_{(m,\vec{k}_\ip,\phi)}(\vec{r}) = 1 + \tfrac{m}{2} \cos\ab(2\pi \vec{k}_\ip \cdot \vec{r} + \phi)
            \]}
            \pause % 2->3
            \only<1-11>{\[
                \F{\ip_{(m,\vec{k}_\ip,\phi)}} =  \frac{me^{-\imag \phi}}{4}
                    \textcolor{Orange}{\dirac_{-\vec{k}_\ip}(\vec{k})}
                    + \textcolor{Orange}{\dirac(\vec{k})} + 
                    \frac{me^{\imag \phi}}{4} \textcolor{Orange}{\dirac_{+\vec{k}_\ip}(\vec{k})}
            \]}
            \pause\pause\pause 
            \only<1-11>{
                \alt<7->{
                    \[
                        \textcolor{Orange}{\text{image formation}}\ldots\begin{cases}
                    \sensed_\text{\acs{SIM}}(\vec{r}) = (\source(\vec{r}) \cdot \ip(\vec{r})) * \psf(\vec{r}) \\
                    \Fsensed_{\text{\acs{SIM}}}(\vec{k}) = (\Fsource(\vec{k}) * \Fip(\vec{k})) \cdot \otf(\vec{k})
                        \end{cases}
                    \] 
                }{
                \[
                    \sensed_\text{\acs{SIM}}(\vec{r}) = (\source(\vec{r}) \cdot \ip(\vec{r})) * \psf(\vec{r})
                \] 
                \pause
                \[
                    \Fsensed_{\text{\acs{SIM}}}(\vec{k}) = (\Fsource(\vec{k}) * \Fip(\vec{k})) \cdot \otf(\vec{k})
                \] 
            }
            }
            \pause\pause\pause
            \[
                \Fsensed_{\text{\acs{SIM}}}(\vec{k}) =  \bigg(\frac{me^{-i \phi}}{4}
                \alt<9>{\underbrace{\Fsource(\vec{k} + \vec{k}_\ip)}_{\textcolor{red}{C_{- \vec{k}_\ip}(\vec{k})}}}{\textcolor{Orange}{\Fsource(\vec{k} + \vec{k}_\ip)}}
                    + 
                \alt<9>{\underbrace{\Fsource(\vec{k})}_{\textcolor{green}{C_{0}(\vec{k})}}}{\textcolor{Orange}{\Fsource(\vec{k})}}
                + \frac{me^{i \phi}}{4} 
                \alt<9>{\underbrace{\Fsource(\vec{k}-\vec{k}_\ip)}_{\textcolor{blue}{C_{+ \vec{k}_\ip}(\vec{k})}}}{\textcolor{Orange}{\Fsource(\vec{k}-\vec{k}_\ip)}}
                \bigg)\cdot\otf(\vec{k})
                \] 
            \pause\pause\pause\pause
        \item in the simplest form, \(3\) orientations and \(3\) phases are used for one \ac{HR} image \pause
            \color{gray}{(9 \ac{LR} images produce \(1\) \ac{HR} image)}
        \end{itemize}
    }
